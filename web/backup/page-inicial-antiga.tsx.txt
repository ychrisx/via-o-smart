const statusClass = (value: string) => {
  const v = value.toLowerCase();
  if (v.includes("atras") || v.includes("vencid") || v.includes("abaixo") || v.includes("inativo") || v.includes("atenção") || v.includes("manutenção")) return "status status-red";
  if (v.includes("vencer") || v.includes("pendente") || v.includes("aguardando") || v.includes("incompleto") || v.includes("análise")) return "status status-amber";
  return "status status-green";
};

function Logo() {
  return <div className="brand" aria-label="Viação Smart"><span>Smart</span></div>;
}

function StatusBadge({ children }: { children: string }) {
  return <span className={statusClass(children)}><i />{children}</span>;
}

function HealthCount({ good, warn, bad }: { good: number; warn: number; bad: number }) {
  return <span className="health-count"><b><i className="green-dot" />{good}</b><b><i className="amber-dot" />{warn}</b><b><i className="red-dot" />{bad}</b></span>;
}

function Table({ rows, onOpen }: { rows: Row[]; onOpen: (row: Row) => void }) {
  if (!rows.length) return <div className="empty-state">Nenhum cadastro encontrado.</div>;
  const headers = Object.keys(rows[0]);
  return <div className="table-scroll"><table className="data-table"><thead><tr>{headers.map((header) => <th key={header}>{header}</th>)}<th className="action-heading">Dados gerais</th></tr></thead><tbody>{rows.map((row, i) => <tr key={`${Object.values(row)[0]}-${i}`} onClick={() => onOpen(row)}>{headers.map((header) => <td key={header}>{header === "situacao" || header === "situação" || header === "cadastro" ? <StatusBadge>{row[header]}</StatusBadge> : row[header]}</td>)}<td><button className="open-link" onClick={(event) => { event.stopPropagation(); onOpen(row); }}>Abrir</button></td></tr>)}</tbody></table></div>;
}
